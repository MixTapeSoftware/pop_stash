# Step 2: Passwordless Authentication

## Objective

Modify the phx.gen.auth output to implement passwordless magic link authentication. Create registration flow that atomically creates org + user + owner membership. Create login flow with email-only magic link.

This step is orthogonal to Steps 1/3/4 -- it only touches the auth system (Accounts context, User schema, UserToken, session controller, auth LiveViews).

## Prerequisites

- Step 0 completed (users, users_tokens, organizations tables exist)
- phx.gen.auth has been run and output exists

### 3. Update Accounts context

**File**: `/workspace/lib/pop_stash/accounts.ex`

```elixir
alias PopStash.Organizations.Organization
alias PopStash.Organizations.OrgMember

@doc """
Registers a new user with their first organization, atomically.

Creates org, user, owner membership, and sets selected_org_id.
Returns {:ok, %{user: user, org: org}} or {:error, changeset_or_reason}.
"""
def register_user_with_org(attrs) do
  Repo.transact(fn ->
    org_name = attrs["org_name"] || attrs[:org_name]
    email = attrs["email"] || attrs[:email]
    slug = Organization.generate_slug(org_name)

    with {:ok, org} <-
           %Organization{}
           |> org_changeset(%{name: org_name, slug: slug})
           |> Repo.insert(),
         {:ok, user} <-
           %User{}
           |> registration_changeset(%{email: email})
           |> Repo.insert(),
         {:ok, _member} <-
           %OrgMember{}
           |> owner_changeset(%{org_id: org.id, user_id: user.id, role: "owner"})
           |> Repo.insert(),
         {:ok, user} <-
           user
           |> Ecto.Changeset.change(selected_org_id: org.id)
           |> Repo.update() do
      {:ok, %{user: user, org: org}}
    end
  end)
end

def deliver_login_link(user, url_fun) when is_function(url_fun, 1) do
  {encoded_token, user_token} = UserToken.build_email_token(user, "login")
  Repo.insert!(user_token)
  UserNotifier.deliver_login_link(user, url_fun.(encoded_token))
end

def verify_login_token(token) do
  with {:ok, query} <- UserToken.verify_email_token_query(token, "login"),
       %User{} = user <- Repo.one(query) do
    # Delete all login tokens for this user (single use)
    Repo.delete_all(UserToken.by_user_and_contexts_query(user, ["login"]))
    {:ok, user}
  else
    _ -> :error
  end
end

def select_org(user, org_id) do
  user
  |> Ecto.Changeset.change(selected_org_id: org_id)
  |> Repo.update()
end

# Changesets (in context per project guidelines)

defp registration_changeset(user, attrs) do
  user
  |> Ecto.Changeset.cast(attrs, [:email])
  |> Ecto.Changeset.validate_required([:email])
  |> Ecto.Changeset.validate_format(:email, ~r/^[^\s]+@[^\s]+$/, message: "must have @ sign and no spaces")
  |> Ecto.Changeset.validate_length(:email, max: 160)
  |> Ecto.Changeset.unsafe_validate_unique(:email, Repo)
  |> Ecto.Changeset.unique_constraint(:email)
end

# No unsafe_validate_unique on slug -- uniqueness only disclosed on submit
defp org_changeset(org, attrs) do
  org
  |> Ecto.Changeset.cast(attrs, [:name, :slug])
  |> Ecto.Changeset.validate_required([:name, :slug])
  |> Ecto.Changeset.validate_length(:name, min: 1, max: 255)
  |> Ecto.Changeset.validate_format(:slug, ~r/^[a-z0-9-]+$/, message: "must contain only lowercase letters, numbers, and hyphens")
  |> Ecto.Changeset.unique_constraint(:slug)
end

defp owner_changeset(member, attrs) do
  member
  |> Ecto.Changeset.cast(attrs, [:org_id, :user_id, :role])
  |> Ecto.Changeset.validate_required([:org_id, :user_id, :role])
  |> Ecto.Changeset.validate_inclusion(:role, ~w(owner member))
  |> Ecto.Changeset.unique_constraint([:org_id, :user_id])
end
```

### 4. Create UserNotifier

**File**: `/workspace/lib/pop_stash/accounts/user_notifier.ex`

```elixir
defmodule PopStash.Accounts.UserNotifier do
  import Swoosh.Email

  alias PopStash.Mailer

  defp deliver(recipient, subject, body) do
    email =
      new()
      |> to(recipient)
      |> from({"PopStash", "noreply@popstash.dev"})
      |> subject(subject)
      |> text_body(body)

    with {:ok, _metadata} <- Mailer.deliver(email) do
      {:ok, email}
    end
  end

  def deliver_login_link(user, url) do
    deliver(user.email, "Your PopStash login link", """
    Hi #{user.email},

    Click the link below to sign in to PopStash:

    #{url}

    This link expires in 1 hour.

    If you didn't request this, ignore this email.
    """)
  end
end
```

defmodule PopStashWeb.UserLoginLive do
  use PopStashWeb, :live_view

  alias PopStash.Accounts

  def mount(_params, _session, socket) do
    email = Phoenix.Flash.get(socket.assigns.flash, :email)
    form = to_form(%{"email" => email}, as: "user")
    {:ok, assign(socket, form: form, page_title: "Sign In")}
  end

  def handle_event("send_link", %{"user" => %{"email" => email}}, socket) do
    if user = Accounts.get_user_by_email(email) do
      Accounts.deliver_login_link(user, &url(~p"/users/log-in/#{&1}"))
    end

    # Always show same message (don't reveal if user exists)
    {:noreply,
     socket
     |> put_flash(:info, "If your email is registered, you will receive a login link shortly.")
     |> assign(form: to_form(%{"email" => ""}, as: "user"))}
  end

  def render(assigns) do
    ~H"""
    <Layouts.app flash={@flash}>
      <div class="mx-auto max-w-sm">
        <.header class="text-center">
          Sign in to PopStash
          <:subtitle>We will send a magic link to your email</:subtitle>
        </.header>

        <.simple_form for={@form} id="login-form" phx-submit="send_link">
          <.input field={@form[:email]} type="email" label="Email" required />
          <:actions>
            <.button phx-disable-with="Sending..." class="w-full">
              Send login link
            </.button>
          </:actions>
        </.simple_form>

        <p class="text-center text-sm mt-4">
          Don't have an account?
          <.link navigate={~p"/users/register"} class="font-semibold text-violet-600 hover:underline">

      </div>

defmodule PopStashWeb.UserRegistrationLive do
  use PopStashWeb, :live_view

  alias PopStash.Accounts

  def mount(_params, _session, socket) do
    form = to_form(%{"email" => "", "org_name" => ""}, as: "registration")
    {:ok, assign(socket, form: form, page_title: "Register")}
  end

  def handle_event("register", %{"registration" => params}, socket) do
    case Accounts.register_user_with_org(params) do
      {:ok, %{user: user}} ->
        Accounts.deliver_login_link(user, &url(~p"/users/log-in/#{&1}"))

        {:noreply,
         socket
         |> put_flash(:info, "Account created! Check your email for a login link.")
         |> assign(form: to_form(%{"email" => "", "org_name" => ""}, as: "registration"))}

      {:error, %Ecto.Changeset{} = changeset} ->
        {:noreply, assign(socket, form: to_form(changeset, as: "registration"))}

      {:error, _reason} ->
        {:noreply,
         socket
         |> put_flash(:error, "Registration failed. Please try again.")
         |> assign(form: to_form(%{"email" => "", "org_name" => ""}, as: "registration"))}
    end
  end

  def render(assigns) do
    ~H"""
    <Layouts.app flash={@flash}>
      <div class="mx-auto max-w-sm">
        <.header class="text-center">
          Create your PopStash account
          <:subtitle>Set up your workspace in seconds</:subtitle>
        </.header>

        <.simple_form for={@form} id="registration-form" phx-submit="register">
          <.input field={@form[:email]} type="email" label="Email" required />
          <.input field={@form[:org_name]} type="text" label="Organization Name" required />
          <:actions>
            <.button phx-disable-with="Creating account..." class="w-full">
              Create account
            </.button>
          </:actions>

def verify_token(conn, %{"token" => token}) do
  case Accounts.verify_login_token(token) do
    {:ok, user} ->
      conn
      |> put_flash(:info, "Welcome back!")
      |> UserAuth.log_in_user(user)

    :error ->
      conn
      |> put_flash(:error, "Login link is invalid or has expired.")
      |> redirect(to: ~p"/users/log-in")
  end
end

defmodule PopStash.AccountsTest do
  use PopStash.DataCase

  alias PopStash.Accounts
  alias PopStash.Organizations.Organization

  describe "register_user_with_org/1" do
    test "creates user, org, and owner membership atomically" do
      assert {:ok, %{user: user, org: org}} =
               Accounts.register_user_with_org(%{
                 "email" => "test@example.com",
                 "org_name" => "Acme Corp"
               })

      assert user.email == "test@example.com"
      assert user.selected_org_id == org.id
      assert org.name == "Acme Corp"
      assert org.slug == "acme-corp"

      # Verify membership
      member = Repo.get_by!(PopStash.Organizations.OrgMember,
        org_id: org.id, user_id: user.id)
      assert member.role == "owner"
    end

    test "returns error for duplicate email" do
      Accounts.register_user_with_org(%{
        "email" => "dupe@example.com",
        "org_name" => "Org 1"
      })

      assert {:error, changeset} =
               Accounts.register_user_with_org(%{
                 "email" => "dupe@example.com",
                 "org_name" => "Org 2"
               })

      assert %{email: _} = errors_on(changeset)
    end

    test "returns error for duplicate org slug" do
      Accounts.register_user_with_org(%{
        "email" => "user1@example.com",
        "org_name" => "Acme Corp"
      })

      assert {:error, changeset} =
               Accounts.register_user_with_org(%{
                 "email" => "user2@example.com",
                 "org_name" => "Acme Corp"
               })

      assert %{slug: _} = errors_on(changeset)
    end

    test "returns error for invalid email" do
      assert {:error, changeset} =
               Accounts.register_user_with_org(%{
                 "email" => "invalid",
                 "org_name" => "Test Org"
               })

      assert %{email: _} = errors_on(changeset)
    end
  end

  describe "verify_login_token/1" do
    test "returns user for valid token" do
      %{user: user} = setup_org_context()

      {token, _} = PopStash.Accounts.UserToken.build_email_token(user, "login")
      Repo.insert!(%PopStash.Accounts.UserToken{
        user_id: user.id,
        token: :crypto.hash(:sha256, Base.url_decode64!(token, padding: false)),
        context: "login",
        sent_to: user.email
      })

      assert {:ok, found_user} = Accounts.verify_login_token(token)
      assert found_user.id == user.id
    end

    test "returns error for invalid token" do
      assert :error = Accounts.verify_login_token("invalid-token")
    end
  end

  describe "select_org/2" do
    test "updates user selected_org_id" do
      %{user: user} = setup_org_context()
      org2 = organization_fixture()

      assert {:ok, updated} = Accounts.select_org(user, org2.id)
      assert updated.selected_org_id == org2.id
    end
  end
end

## Important Notes

- The `Layouts.app` wrapper in LiveView templates follows Phoenix 1.8 conventions. The `current_scope` assign is NOT available yet in this step -- that comes in Step 5 with OrgPlug. These LiveViews render on unauthenticated routes and do not need `current_scope`.
- The `UserAuth` module generated by phx.gen.auth handles session management. We keep its `log_in_user/2`, `log_out_user/1`, `fetch_current_user/2`, `require_authenticated_user/2`, and `redirect_if_user_is_authenticated/2` functions.
- Remove all password-related functions from UserAuth (`require_authenticated_user` should NOT check for password confirmation).
